"""Rough raster preview of the generated PPTX (approximates PowerPoint layout).

Usage: python3 preview.py deck.pptx outdir [dpi]
"""
import os
import sys
import glob

from PIL import Image, ImageDraw, ImageFont
from pptx import Presentation
from pptx.util import Emu
from pptx.enum.shapes import MSO_SHAPE_TYPE
from pptx.enum.text import PP_ALIGN, MSO_ANCHOR

HERE = os.path.dirname(os.path.abspath(__file__))
REG = os.path.join(HERE, "NotoSansSC-Regular.otf")
BOLD = os.path.join(HERE, "NotoSansSC-Bold.otf")

_font_cache = {}


def get_font(size_px, bold):
    key = (int(size_px), bold)
    if key not in _font_cache:
        path = BOLD if (bold and os.path.exists(BOLD)) else REG
        try:
            _font_cache[key] = ImageFont.truetype(path, size_px)
        except Exception:
            _font_cache[key] = ImageFont.truetype(REG, size_px)
    return _font_cache[key]


def hexcolor(color_obj, default="#000000"):
    try:
        if color_obj and color_obj.type is not None:
            return "#" + str(color_obj.rgb)
    except Exception:
        pass
    return default


def wrap(text, font, max_w, draw):
    lines, cur = [], ""
    for ch in text:
        if ch == "\n":
            lines.append(cur)
            cur = ""
            continue
        if draw.textlength(cur + ch, font=font) <= max_w or not cur:
            cur += ch
        else:
            lines.append(cur)
            cur = ch
    lines.append(cur)
    return lines


def draw_text_frame(draw, shape, dpi):
    tf = shape.text_frame
    k = dpi / 72.0
    left = shape.left / 914400 * dpi
    top = shape.top / 914400 * dpi
    width = shape.width / 914400 * dpi
    height = shape.height / 914400 * dpi
    ml = (tf.margin_left or 0) / 914400 * dpi
    mr = (tf.margin_right or 0) / 914400 * dpi
    mt = (tf.margin_top or 0) / 914400 * dpi
    avail = max(10, width - ml - mr)

    blocks = []
    for p in tf.paragraphs:
        runs = [r for r in p.runs if r.text]
        if not runs:
            blocks.append(([], p, 0, 0))
            continue
        size = (runs[0].font.size.pt if runs[0].font.size else 18.0)
        ls = p.line_spacing if p.line_spacing else 1.2
        ls = ls if isinstance(ls, float) else 1.2
        body = "".join(r.text for r in runs)
        font = get_font(int(round(size * k)), bool(runs[0].font.bold))
        lines = wrap(body, font, avail, draw)
        color = hexcolor(runs[0].font.color, "#23313D")
        line_h = size * 1.2 * ls * k
        blocks.append((lines, p, line_h, font, color, runs, size))

    total = sum(len(b[0]) * (b[2] if len(b) > 2 else 0) for b in blocks)
    total += sum((p.space_after.pt * k if p.space_after else 0) for _, p, *_ in blocks)
    y = top + mt
    if tf.vertical_anchor == MSO_ANCHOR.MIDDLE:
        y = top + (height - total) / 2.0
    for blk in blocks:
        lines, p, line_h, font, color, runs, size = blk
        for ln in lines:
            align = p.alignment
            w = draw.textlength(ln, font=font)
            if align == PP_ALIGN.CENTER:
                x = left + ml + (avail - w) / 2
            elif align == PP_ALIGN.RIGHT:
                x = left + ml + avail - w
            else:
                x = left + ml
            draw.text((x, y), ln, font=font, fill=color,
                      stroke_width=1 if (runs and runs[0].font.bold) else 0,
                      stroke_fill=color)
            y += line_h
        if p.space_after:
            y += p.space_after.pt * k


def draw_table(draw, shape, dpi):
    tbl = shape.table
    x = shape.left / 914400 * dpi
    y = shape.top / 914400 * dpi
    for i, row in enumerate(tbl.rows):
        h = row.height / 914400 * dpi
        cx = x
        for j, cell in enumerate(row.cells):
            w = tbl.columns[j].width / 914400 * dpi
            col = "#FFFFFF"
            try:
                if cell.fill.type is not None:
                    col = hexcolor(cell.fill.fore_color, "#FFFFFF")
            except Exception:
                pass
            draw.rectangle([cx, y, cx + w, y + h], fill=col)
            txt = cell.text_frame.text
            if txt:
                p = cell.text_frame.paragraphs[0]
                size = 13.5
                bold = False
                tcol = "#23313D"
                if p.runs:
                    if p.runs[0].font.size:
                        size = p.runs[0].font.size.pt
                    bold = bool(p.runs[0].font.bold)
                    tcol = hexcolor(p.runs[0].font.color, "#23313D")
                font = get_font(int(round(size * dpi / 72.0)), bold)
                tw = draw.textlength(txt, font=font)
                if p.alignment == PP_ALIGN.CENTER:
                    tx = cx + (w - tw) / 2
                elif p.alignment == PP_ALIGN.RIGHT:
                    tx = cx + w - tw - 8
                else:
                    tx = cx + 0.12 * dpi
                draw.text((tx, y + (h - size * dpi / 72.0) / 2), txt, font=font,
                          fill=tcol,
                          stroke_width=1 if bold else 0, stroke_fill=tcol)
            cx += w
        y += h


def render(pptx_path, outdir, dpi=150):
    os.makedirs(outdir, exist_ok=True)
    prs = Presentation(pptx_path)
    W = int(prs.slide_width / 914400 * dpi)
    H = int(prs.slide_height / 914400 * dpi)
    paths = []
    for idx, slide in enumerate(prs.slides, start=1):
        img = Image.new("RGB", (W, H), "white")
        draw = ImageDraw.Draw(img)
        for shape in slide.shapes:
            st = shape.shape_type
            if getattr(shape, "has_table", False) and shape.has_table:
                draw_table(draw, shape, dpi)
                continue
            if st == MSO_SHAPE_TYPE.PICTURE:
                pic = Image.open(shape.image.blob and
                                 __import__("io").BytesIO(shape.image.blob))
                pic = pic.convert("RGBA")
                tw = int(shape.width / 914400 * dpi)
                th = int(shape.height / 914400 * dpi)
                pic = pic.resize((max(1, tw), max(1, th)), Image.LANCZOS)
                img.paste(pic, (int(shape.left / 914400 * dpi),
                                int(shape.top / 914400 * dpi)), pic)
                continue
            # autoshape / textbox: fill first
            try:
                if st == MSO_SHAPE_TYPE.AUTO_SHAPE:
                    x = shape.left / 914400 * dpi
                    y = shape.top / 914400 * dpi
                    w = shape.width / 914400 * dpi
                    h = shape.height / 914400 * dpi
                    fill = shape.fill
                    col = None
                    if fill.type is not None and str(fill.type) != "BACKGROUND (5)":
                        col = hexcolor(fill.fore_color, "#FFFFFF")
                    name = str(shape.auto_shape_type)
                    if col:
                        if "ROUNDED" in name:
                            try:
                                adj = shape.adjustments[0]
                            except Exception:
                                adj = 0.1
                            r = max(1, adj * min(w, h) / 2)
                            draw.rounded_rectangle([x, y, x + w, y + h],
                                                   radius=r, fill=col)
                        elif "OVAL" in name:
                            draw.ellipse([x, y, x + w, y + h], fill=col)
                        else:
                            draw.rectangle([x, y, x + w, y + h], fill=col)
            except Exception as e:
                print("  shape err", e)
            if shape.has_text_frame and shape.text_frame.text.strip():
                draw_text_frame(draw, shape, dpi)
        p = os.path.join(outdir, f"slide{idx:02d}.png")
        img.save(p)
        paths.append(p)
        print("rendered", p, img.size)
    return paths


if __name__ == "__main__":
    deck = sys.argv[1]
    out = sys.argv[2]
    dpi = int(sys.argv[3]) if len(sys.argv) > 3 else 150
    render(deck, out, dpi)

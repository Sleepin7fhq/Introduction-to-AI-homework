# ---------------------------------------------------------------- helpers ---
def rgb(hexstr):
    return RGBColor.from_string(str(hexstr).lstrip("#").upper())


def style_run(run, size=14, bold=False, color=INK, name=CJK):
    f = run.font
    f.size = Pt(size)
    f.bold = bold
    f.color.rgb = rgb(color)
    f.name = name
    rPr = run._r.get_or_add_rPr()
    for tag in ("a:ea", "a:cs"):
        el = rPr.find(qn(tag))
        if el is None:
            el = rPr.makeelement(qn(tag), {})
            rPr.append(el)
        el.set("typeface", name)


def textbox(slide, x, y, w, h, paras, anchor=MSO_ANCHOR.TOP,
            align=PP_ALIGN.LEFT, wrap=True):
    tb = slide.shapes.add_textbox(Inches(x), Inches(y), Inches(w), Inches(h))
    tf = tb.text_frame
    tf.word_wrap = wrap
    tf.vertical_anchor = anchor
    tf.margin_left = tf.margin_right = tf.margin_top = tf.margin_bottom = 0
    for i, spec in enumerate(paras):
        p = tf.paragraphs[0] if i == 0 else tf.add_paragraph()
        p.alignment = spec.get("align", align)
        p.line_spacing = spec.get("line_spacing", 1.32)
        p.space_after = Pt(spec.get("space_after", 0))
        runs = spec["text"] if isinstance(spec["text"], list) else [spec]
        for rspec in runs:
            r = p.add_run()
            r.text = rspec["text"]
            style_run(r, rspec.get("size", spec.get("size", 14)),
                      rspec.get("bold", spec.get("bold", False)),
                      rspec.get("color", spec.get("color", INK)),
                      rspec.get("name", spec.get("name", CJK)))
    return tb


def rect(slide, x, y, w, h, fill, radius=None, line=None, line_w=0.75):
    shp_type = MSO_SHAPE.ROUNDED_RECTANGLE if radius else MSO_SHAPE.RECTANGLE
    sh = slide.shapes.add_shape(shp_type, Inches(x), Inches(y),
                                Inches(w), Inches(h))
    if radius:
        sh.adjustments[0] = min(0.5, radius / (min(w, h) / 2.0))
    if fill:
        sh.fill.solid()
        sh.fill.fore_color.rgb = rgb(fill)
    else:
        sh.fill.background()
    if line:
        sh.line.color.rgb = rgb(line)
        sh.line.width = Pt(line_w)
    else:
        sh.line.fill.background()
    sh.shadow.inherit = False
    return sh


def card(slide, x, y, w, h, fill=LIGHT, accent=None, radius=0.07):
    sh = rect(slide, x, y, w, h, fill, radius=radius)
    if accent:
        rect(slide, x, y, 0.055, h, accent, radius=0.02)
    return sh


def arrow(slide, x, y, w, h, fill=GOLD):
    sh = slide.shapes.add_shape(MSO_SHAPE.RIGHT_ARROW, Inches(x), Inches(y),
                                Inches(w), Inches(h))
    sh.fill.solid()
    sh.fill.fore_color.rgb = rgb(fill)
    sh.line.fill.background()
    sh.shadow.inherit = False
    return sh


def badge(slide, x, y, num, d=0.42, fill=NAVY, color=WHITE, size=15):
    sh = slide.shapes.add_shape(MSO_SHAPE.OVAL, Inches(x), Inches(y),
                                Inches(d), Inches(d))
    sh.fill.solid()
    sh.fill.fore_color.rgb = rgb(fill)
    sh.line.fill.background()
    sh.shadow.inherit = False
    tf = sh.text_frame
    tf.margin_left = tf.margin_right = tf.margin_top = tf.margin_bottom = 0
    tf.vertical_anchor = MSO_ANCHOR.MIDDLE
    p = tf.paragraphs[0]
    p.alignment = PP_ALIGN.CENTER
    r = p.add_run()
    r.text = str(num)
    style_run(r, size, True, color)
    return sh


class F:
    """formula asset placer (sizes derived from the rendered PNG metrics)"""

    def __init__(self, info, dpi=200):
        self.info = info
        self.dpi = dpi

    def size(self, pt):
        i = self.info
        s = pt / i["pt"]
        return i["w"] / self.dpi * s, i["h"] / self.dpi * s

    def fit(self, pt_cap, max_w):
        i = self.info
        return min(pt_cap, i["pt"] * max_w / (i["w"] / self.dpi))

    def at(self, slide, cx, cy, pt):
        w, h = self.size(pt)
        slide.shapes.add_picture(self.info["path"], Inches(cx - w / 2),
                                 Inches(cy - h / 2), Inches(w), Inches(h))
        return w, h

    def left(self, slide, x, cy, pt):
        w, h = self.size(pt)
        slide.shapes.add_picture(self.info["path"], Inches(x),
                                 Inches(cy - h / 2), Inches(w), Inches(h))
        return w, h

    def right(self, slide, x, cy, pt):
        w, h = self.size(pt)
        slide.shapes.add_picture(self.info["path"], Inches(x - w),
                                 Inches(cy - h / 2), Inches(w), Inches(h))
        return w, h

    textbox(slide, M, 0.42, 9.4, 0.62,

    rect(slide, M + 0.02, 1.12, 0.92, 0.05, GOLD)
    textbox(slide, SW - M - 4.2, 0.55, 4.2, 0.4,
            [{"text": tag, "size": 12, "color": GRAY}], align=PP_ALIGN.RIGHT)
    textbox(slide, M, 6.99, 6.0, 0.3,
            [{"text": "取整函数极限 · 夹逼定理", "size": 10.5, "color": GRAY}])
    textbox(slide, SW - M - 1.2, 6.99, 1.2, 0.3,
            [{"text": f"{num:02d}", "size": 10.5, "bold": True, "color": NAVY}],
            align=PP_ALIGN.RIGHT)


def step_card(slide, x, y, w, h, num, title, fill=LIGHT, accent=None,
              badge_fill=NAVY):
    card(slide, x, y, w, h, fill=fill, accent=accent, radius=0.08)
    badge(slide, x + 0.3, y + 0.205, num, fill=badge_fill)
    textbox(slide, x + 0.92, y + 0.28, w - 1.3, 0.4,
            [{"text": title, "size": 16, "bold": True, "color": NAVY}])

def slide_numeric(prs, fx, plot):
    slide = prs.slides.add_slide(prs.slide_layouts[6])
    header(slide, 6, "数值验证与图像", "结论与延伸")

    rows = [("x", "[1/x]", "x·[1/x]"),
            ("0.4", "2", "0.80"),
            ("0.3", "3", "0.90"),
            ("0.1", "10", "1.00"),
            ("?0.1", "?10", "1.00"),
            ("?0.3", "?4", "1.20"),
            ("?0.4", "?3", "1.20")]
    x0, y0, tw, rh = M, 1.42, 6.05, 0.5
    shape = slide.shapes.add_table(len(rows), 3, Inches(x0), Inches(y0),
                                   Inches(tw), Inches(rh * len(rows)))
    table = shape.table
    table.first_row = False
    table.horz_banding = False
    # switch to the built-in "No Style, No Grid" so only our own fills show
    style_id = table._tbl.tblPr.find(qn("a:tableStyleId"))
    if style_id is not None:
        style_id.text = "{2D5ABB26-0587-4C30-8999-92F81FD0307C}"
    for c, w in zip(table.columns, (2.0, 1.95, 2.1)):
        c.width = Inches(w)
    for i, row in enumerate(rows):
        table.rows[i].height = Inches(rh)
        for j, val in enumerate(row):
            cell = table.cell(i, j)
            cell.margin_left = Inches(0.12)
            cell.margin_right = Inches(0.12)
            cell.margin_top = Inches(0.02)
            cell.margin_bottom = Inches(0.02)
            cell.vertical_anchor = MSO_ANCHOR.MIDDLE
            cell.fill.solid()
            cell.fill.fore_color.rgb = rgb(NAVY if i == 0 else
                                           (WHITE if i % 2 else LIGHT))
            p = cell.text_frame.paragraphs[0]
            p.alignment = PP_ALIGN.LEFT if j == 0 else PP_ALIGN.CENTER
            r = p.add_run()
            r.text = val
            if i == 0:
                style_run(r, 13.5, True, WHITE)
            elif j == 2:
                style_run(r, 13.5, True, CRIMSON)
            else:
                style_run(r, 13.5, False, INK)
    textbox(slide, x0 + 0.05, y0 + rh * len(rows) + 0.18, tw - 0.1, 0.65,
            [{"text": "表中每个值都落在 1 ± |x| 之间：右侧 1 ? x < y ≤ 1，"
                      "左侧 1 ≤ y < 1 ? x。", "size": 12.5, "color": MUTED}])

    slide.shapes.add_picture(plot["path"], Inches(7.28), Inches(1.42),
                             width=Inches(5.25))
    textbox(slide, 7.28, 5.12, 5.25, 0.6,
            [{"text": "曲线随 x → 0 被夹在逐渐收拢的带形区域内，并趋于点 (0, 1)。",
              "size": 12.5, "color": MUTED}])

    y = 5.86
    card(slide, M, y, CW, 1.0, fill=NAVY, accent=GOLD, radius=0.08)
    fx["final_white"].left(slide, M + 0.42, y + 0.5, 25)
    textbox(slide, M + 3.3, y + 0.19, 3.9, 0.62,
            [{"text": "结论：左右极限相等，故原极限为 1。",
              "size": 13.5, "bold": True, "color": "EAF1F8"}],
            anchor=MSO_ANCHOR.MIDDLE)
    textbox(slide, 7.4, y + 0.19, 2.45, 0.62,
            [{"text": "延伸（a ≠ 0）", "size": 12.5, "color": "AFC3D6"}],
            anchor=MSO_ANCHOR.MIDDLE, align=PP_ALIGN.RIGHT)
    fx["gen"].right(slide, SW - M - 0.42, y + 0.5, 20)
    return slide
